/// Конфигурация одного gateway-эндпоинта.
/// Приходит с Dart-стороны как элемент JSON-массива через FFI (transport_start).
#[derive(Debug, Clone, Deserialize, PartialEq, Eq)]
pub struct EndpointConfig {
    pub host: String,
    pub port: u16,
    pub sni: String,
    /// hex-encoded 32-byte X25519 public key of the server
    pub server_pub: String,
    /// hex-encoded 32-byte pre-shared key
    pub psk: String,
}

#[derive(Debug)]
pub enum EndpointConfigError {
    EmptyList,
    TooManyEndpoints(usize),
    InvalidHex { field: &'static str, index: usize },
    WrongHexLength { field: &'static str, index: usize, expected: usize, got: usize },
    MalformedJson(String),
}

impl std::fmt::Display for EndpointConfigError {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            Self::EmptyList => write!(f, "endpoint list is empty"),
            Self::TooManyEndpoints(n) => write!(f, "too many endpoints: {n} (max 16)"),
            Self::InvalidHex { field, index } => {
                write!(f, "invalid hex in field '{field}' at endpoint index {index}")
            }
            Self::WrongHexLength { field, index, expected, got } => {
                write!(
                    f,
                    "field '{field}' at index {index} has wrong length: expected {expected} hex chars, got {got}"
                )
            }
            Self::MalformedJson(msg) => write!(f, "malformed JSON: {msg}"),
        }
    }
}

impl std::error::Error for EndpointConfigError {}

const MAX_ENDPOINTS: usize = 16;
const KEY_HEX_LEN: usize = 64; // 32 bytes -> 64 hex chars

/// Парсит и валидирует JSON-массив EndpointConfig, приходящий из FFI.
pub fn parse_endpoints(json: &str) -> Result<Vec<EndpointConfig>, EndpointConfigError> {
    let endpoints: Vec<EndpointConfig> =
        serde_json::from_str(json).map_err(|e| EndpointConfigError::MalformedJson(e.to_string()))?;

    if endpoints.is_empty() {
        return Err(EndpointConfigError::EmptyList);
    }
    if endpoints.len() > MAX_ENDPOINTS {
        return Err(EndpointConfigError::TooManyEndpoints(endpoints.len()));
    }

    for (i, ep) in endpoints.iter().enumerate() {
        validate_hex_field("server_pub", &ep.server_pub, i)?;
        validate_hex_field("psk", &ep.psk, i)?;
    }

    Ok(endpoints)
}

fn validate_hex_field(field: &'static str, value: &str, index: usize) -> Result<(), EndpointConfigError> {
    if value.len() != KEY_HEX_LEN {
        return Err(EndpointConfigError::WrongHexLength {
            field,
            index,
            expected: KEY_HEX_LEN,
            got: value.len(),
        });
    }
    if !value.chars().all(|c| c.is_ascii_hexdigit()) {
        return Err(EndpointConfigError::InvalidHex { field, index });
    }
    Ok(())
}

//! Multi-endpoint failover для gateway подключения.
//!
//! Принципы (reviewer Phase 1 + AMO_SPEC):
//! - Deterministic priority list (не dynamic discovery)
//! - Failover только по timeout/connection-refused/DNS-fail/TLS-handshake-fail
//!   (НЕ по protocol error mid-handshake — иначе DPI fingerprint)
//! - Cooldown ПЕР-ENDPOINT (не глобальный): минимум 60s на каждый конкретный endpoint
//! - Список endpoints из EndpointConfig (см. endpoint.rs, A1)

use crate::endpoint::EndpointConfig;
use std::sync::Mutex;
use std::time::{Duration, Instant};

/// Минимальный интервал cooldown для ОДНОГО endpoint.
/// Слишком частое переключение = DPI fingerprint (смена IP при неудаче).
pub const FAILOVER_MIN_INTERVAL: Duration = Duration::from_secs(60);

/// Порог подряд идущих failure для конкретного endpoint, после которого
/// делаем попытку переключения на следующий здоровый endpoint.
pub const FAILURE_THRESHOLD: u32 = 3;

#[derive(Debug, Default, Clone)]
struct EndpointState {
    consecutive_failures: u32,
    /// Момент последнего failure для этого endpoint (для per-endpoint cooldown).
    marked_bad_at: Option<Instant>,
}

/// Failover manager: deterministic priority list + PER-ENDPOINT cooldown.
/// Хранит current index и состояние каждого endpoint отдельно. Потокобезопасен через Mutex.
pub struct FailoverManager {
    endpoints: Vec<EndpointConfig>,
    current_idx: Mutex<usize>,
    states: Mutex<Vec<EndpointState>>,
}

impl FailoverManager {
    pub fn new(endpoints: Vec<EndpointConfig>) -> Result<Self, &'static str> {
        if endpoints.is_empty() {
            return Err("endpoints list is empty");
        }
        let states = vec![EndpointState::default(); endpoints.len()];
        Ok(Self {
            endpoints,
            current_idx: Mutex::new(0),
            states: Mutex::new(states),
        })
    }

    /// Возвращает текущий endpoint (клон, чтобы не держать lock снаружи).
    pub fn current_endpoint(&self) -> EndpointConfig {
        let idx = *self.current_idx.lock().unwrap();
        self.endpoints[idx].clone()
    }

    /// Алиас для обратной совместимости с существующими вызовами `.current()`.
    pub fn current(&self) -> EndpointConfig {
        self.current_endpoint()
    }

    pub fn current_index(&self) -> usize {
        *self.current_idx.lock().unwrap()
    }

    pub fn len(&self) -> usize {
        self.endpoints.len()
    }

    /// Регистрирует failure на ТЕКУЩЕМ endpoint (timeout / connection-refused /
    /// DNS-fail / TLS-handshake-fail — НЕ protocol error mid-handshake).
    /// Если счётчик подряд идущих failure достиг порога — пробует переключиться
    /// на следующий endpoint, для которого cooldown уже прошёл.
    /// Возвращает true, если переключение произошло.
    pub fn report_failure(&self) -> bool {
        let idx = *self.current_idx.lock().unwrap();
        let now = Instant::now();

        {
            let mut states = self.states.lock().unwrap();
            states[idx].consecutive_failures += 1;
            states[idx].marked_bad_at = Some(now);
        }

        let should_switch = {
            let states = self.states.lock().unwrap();
            states[idx].consecutive_failures >= FAILURE_THRESHOLD
        };

        if should_switch {
            self.try_switch()
        } else {
            false
        }
    }

    /// Сбрасывает счётчик failure для текущего endpoint (вызывать при успешном подключении).
    pub fn report_success(&self) {
        let idx = *self.current_idx.lock().unwrap();
        let mut states = self.states.lock().unwrap();
        states[idx].consecutive_failures = 0;
        states[idx].marked_bad_at = None;
    }

    /// Пытается переключиться на следующий ЗДОРОВЫЙ endpoint (cooldown которого прошёл).
    /// Обходит список по кругу от текущего +1. Если все endpoints в cooldown —
    /// не переключается (fail-closed на текущем).
    /// Возвращает true, если переключение выполнено.
    pub fn try_switch(&self) -> bool {
        let mut idx = self.current_idx.lock().unwrap();
        let states = self.states.lock().unwrap();
        let len = self.endpoints.len();
        let now = Instant::now();

        for step in 1..=len {
            let candidate = (*idx + step) % len;
            if Self::is_endpoint_available(&states[candidate], now) {
                drop(states);
                *idx = candidate;
                return true;
            }
        }
        false
    }

    /// endpoint доступен, если у него нет failure либо его персональный cooldown прошёл.
    fn is_endpoint_available(state: &EndpointState, now: Instant) -> bool {
        match state.marked_bad_at {
            None => true,
            Some(t) => now.duration_since(t) >= FAILOVER_MIN_INTERVAL,
        }
    }

    /// Разрешено ли сейчас переключение (хотя бы один альтернативный endpoint здоров).
    pub fn can_switch(&self) -> bool {
        let idx = *self.current_idx.lock().unwrap();
        let states = self.states.lock().unwrap();
        let len = self.endpoints.len();
        let now = Instant::now();
        (1..len).any(|step| {
            let candidate = (idx + step) % len;
            Self::is_endpoint_available(&states[candidate], now)
        })
    }
}

/// Локальный SOCKS5-прокси (127.0.0.1).
///
/// Схема работы:
///   GeckoView / WebView  →  SOCKS5 (127.0.0.1:порт)
///                              │
///                        proxy.rs (этот файл)
///                              │
///                        FailoverManager (endpoint selection)
///                              │
///                        protocol.rs (FrameCodec)
///                              │
///                        tls.rs → удалённый gateway

use std::net::SocketAddr;
use std::sync::Arc;
use tokio::io::{AsyncReadExt, AsyncWriteExt};
use tokio::net::{TcpListener, TcpStream};
use tracing::{debug, info, warn};

use crate::error::TransportError;
use crate::failover::FailoverManager;
use crate::endpoint::EndpointConfig;
use crate::protocol::FrameCodec;
use crate::rate_limiter::RateLimiter;
use crate::tls::TlsClient;

const SOCKS_VERSION: u8 = 0x05;
const NO_AUTH: u8       = 0x00;
const CMD_CONNECT: u8   = 0x01;
const ATYP_IPV4: u8     = 0x01;
const ATYP_DOMAIN: u8   = 0x03;
const ATYP_IPV6: u8     = 0x04;
const REP_SUCCESS: u8   = 0x00;
const REP_FAILURE: u8   = 0x01;

pub struct Socks5Proxy {
    bind_addr: SocketAddr,
    failover: Arc<FailoverManager>,
    key: [u8; 32],
    sni_pool: Vec<String>,
    rate_limiter: Option<Arc<RateLimiter>>,
}

impl Socks5Proxy {
    pub fn new(
        bind_addr: SocketAddr,
        failover: Arc<FailoverManager>,
        key: [u8; 32],
        sni_pool: Vec<String>,
    ) -> Self {
        Self {
            bind_addr,
            failover,
            key,
            sni_pool,
            rate_limiter: None,
        }
    }

    pub fn with_rate_limit(mut self, bytes_per_second: u64) -> Self {
        self.rate_limiter = Some(Arc::new(RateLimiter::new(bytes_per_second)));
        self
    }

    /// Запускает прокси-сервер. Блокирует до ошибки listener.
    pub async fn run(self) -> Result<(), TransportError> {
        let listener = TcpListener::bind(self.bind_addr).await?;
        info!("SOCKS5 proxy listening on {}", self.bind_addr);

        let _key = self.key;
        let sni_pool = self.sni_pool;
        let rate_limiter = self.rate_limiter;
        let failover = self.failover;

        loop {
            let (client, peer) = listener.accept().await?;
            debug!("new connection from {}", peer);

            let sni = sni_pool.clone();
            let rl = rate_limiter.clone();
            let fo = failover.clone();
            tokio::spawn(async move {
                if let Err(e) = handle_connection(client, fo, sni, rl).await {
                    warn!("connection {} error: {}", peer, e);
                }
            });
        }
    }
}

/// Устанавливает TLS-соединение с текущим endpoint из FailoverManager.
/// При сетевой ошибке (timeout/refused/DNS/TLS-handshake) — report_failure()
/// (может триггернуть try_switch внутри) и retry на следующем endpoint.
/// Максимум `len` попыток (один полный круг по списку endpoints).
async fn connect_with_failover(
    failover: &Arc<FailoverManager>,
    sni_pool: &[String],
) -> Result<(tokio_rustls::client::TlsStream<TcpStream>, EndpointConfig), TransportError> {
    let attempts = failover.len();
    let mut last_err: Option<TransportError> = None;

    for _ in 0..attempts {
        let endpoint = failover.current_endpoint();
        let mut tls_client = TlsClient::new(sni_pool.to_vec())?;

        match tls_client.connect(&endpoint.host, endpoint.port).await {
            Ok(stream) => {
                failover.report_success();
                return Ok((stream, endpoint));
            }
            Err(e) => {
                warn!(
                    "connect failed for endpoint {}:{} — {} (reporting failure, may trigger failover)",
                    endpoint.host, endpoint.port, e
                );
                // Failover ТОЛЬКО по: timeout / connection refused / DNS fail / TLS handshake fail.
                // Все эти случаи происходят внутри tls_client.connect(), до steal-handshake —
                // поэтому здесь безопасно report_failure() для любой ошибки этого этапа.
                failover.report_failure();
                last_err = Some(e);
            }
        }
    }

    Err(last_err.unwrap_or_else(|| {
        TransportError::Protocol("no endpoints available (all in cooldown)".into())
    }))
}

async fn handle_connection(
    mut client: TcpStream,
    failover: Arc<FailoverManager>,
    sni_pool: Vec<String>,
    rate_limiter: Option<Arc<RateLimiter>>,
) -> Result<(), TransportError> {
    let mut header = [0u8; 2];
    client.read_exact(&mut header).await?;

    if header[0] != SOCKS_VERSION {
        return Err(TransportError::Protocol("not SOCKS5".into()));
    }

    let nmethods = header[1] as usize;
    let mut methods = vec![0u8; nmethods];
    client.read_exact(&mut methods).await?;

    if !methods.contains(&NO_AUTH) {
        client.write_all(&[SOCKS_VERSION, 0xFF]).await?;
        return Err(TransportError::Protocol("no acceptable auth method".into()));
    }
    client.write_all(&[SOCKS_VERSION, NO_AUTH]).await?;

    let mut req = [0u8; 4];
    client.read_exact(&mut req).await?;

    if req[0] != SOCKS_VERSION {
        return Err(TransportError::Protocol("bad SOCKS version in request".into()));
    }
    if req[1] != CMD_CONNECT {
        client
            .write_all(&[SOCKS_VERSION, 0x07, 0x00, ATYP_IPV4, 0, 0, 0, 0, 0, 0])
            .await?;
        return Err(TransportError::Protocol("only CONNECT supported".into()));
    }

    let target_host = match req[3] {
        ATYP_IPV4 => {
            let mut ip = [0u8; 4];
            client.read_exact(&mut ip).await?;
            std::net::Ipv4Addr::from(ip).to_string()
        }
        ATYP_DOMAIN => {
            let mut len = [0u8; 1];
            client.read_exact(&mut len).await?;
            let mut domain = vec![0u8; len[0] as usize];
            client.read_exact(&mut domain).await?;
            String::from_utf8(domain)
                .map_err(|_| TransportError::Protocol("invalid domain encoding".into()))?
        }
        ATYP_IPV6 => {
            let mut ip = [0u8; 16];
            client.read_exact(&mut ip).await?;
            std::net::Ipv6Addr::from(ip).to_string()
        }
        t => {
            return Err(TransportError::Protocol(format!("unknown atyp: {}", t)));
        }
    };

    let mut port_bytes = [0u8; 2];
    client.read_exact(&mut port_bytes).await?;
    let target_port = u16::from_be_bytes(port_bytes);

    debug!("CONNECT {}:{}", target_host, target_port);

    // ── 1. Установка TLS до gateway, с failover при сетевых ошибках ──
    let (mut tls_stream, endpoint) = connect_with_failover(&failover, &sni_pool).await?;

    // ── 2. Steal-handshake. Ошибки здесь — protocol error mid-handshake,
    //       НЕ вызывают report_failure() (риск DPI-фингерпринта). ──
    let server_public = hex_to_32(&endpoint.server_pub)?;
    let client_psk = hex_to_32(&endpoint.psk)?;
    let client_auth =
        crate::steal::client_handshake(&mut tls_stream, &server_public, &client_psk).await?;

    let codec = FrameCodec::new(&client_auth.session_key, client_auth.c2s_prefix, client_auth.s2c_prefix, 0);

    // Первый фрейм — CONNECT-запрос: "host:port"
    let connect_msg = format!("CONNECT {}:{}", target_host, target_port);
    codec.write_frame(&mut tls_stream, connect_msg.as_bytes()).await?;

    // Ответ gateway
    let ack = codec.read_frame(&mut tls_stream).await?;
    if ack != b"OK" {
        let msg = String::from_utf8_lossy(&ack).to_string();
        client
            .write_all(&[SOCKS_VERSION, REP_FAILURE, 0x00, ATYP_IPV4, 0, 0, 0, 0, 0, 0])
            .await?;
        return Err(TransportError::Protocol(format!("gateway refused: {}", msg)));
    }

    // Сообщаем клиенту: соединение установлено
    client
        .write_all(&[SOCKS_VERSION, REP_SUCCESS, 0x00, ATYP_IPV4, 0, 0, 0, 0, 0, 0])
        .await?;

    // ── 3. Прозрачный двунаправленный туннель ──
    relay(client, tls_stream, codec, rate_limiter).await
}

fn hex_to_32(hex: &str) -> Result<[u8; 32], TransportError> {
    let bytes = hex_decode(hex)
        .map_err(|_| TransportError::Protocol("invalid hex key".into()))?;
    if bytes.len() != 32 {
        return Err(TransportError::Protocol("key must be 32 bytes".into()));
    }
    let mut out = [0u8; 32];
    out.copy_from_slice(&bytes);
    Ok(out)
}

fn hex_decode(s: &str) -> Result<Vec<u8>, ()> {
    if s.len() % 2 != 0 {
        return Err(());
    }
    (0..s.len())
        .step_by(2)
        .map(|i| u8::from_str_radix(&s[i..i + 2], 16).map_err(|_| ()))
        .collect()
}

/// Проксирует данные между клиентом и gateway в обоих направлениях.
async fn relay(
    mut client: TcpStream,
    mut gateway: tokio_rustls::client::TlsStream<tokio::net::TcpStream>,
    codec: FrameCodec,
    rate_limiter: Option<Arc<RateLimiter>>,
) -> Result<(), TransportError> {
    let mut client_buf = vec![0u8; 8192];

    loop {
        tokio::select! {
            n = client.read(&mut client_buf) => {
                let n = n?;
                if n == 0 {
                    debug!("client closed connection");
                    return Ok(());
                }
                if let Some(ref rl) = rate_limiter {
                    rl.wait_for_bytes_async(n as u64).await;
                }
                codec.write_frame(&mut gateway, &client_buf[..n]).await?;
            }

            frame = codec.read_frame(&mut gateway) => {
                match frame {
                    Ok(data) => {
                        if data.len() >= crate::protocol::REKEY_INIT_MAGIC.len()
                            && &data[..crate::protocol::REKEY_INIT_MAGIC.len()]
                                == crate::protocol::REKEY_INIT_MAGIC
                        {
                            debug!("received REKEY_INIT from gateway — performing rekey");
                            let magic_len = crate::protocol::REKEY_INIT_MAGIC.len();
                            if data.len() < magic_len + 1 + 32 {
                                return Err(TransportError::Protocol("REKEY_INIT too short".into()));
                            }
                            let new_kid = data[magic_len];
                            let mut new_key = [0u8; 32];
                            new_key.copy_from_slice(&data[magic_len + 1..magic_len + 1 + 32]);

                            let mut ack = Vec::with_capacity(crate::protocol::REKEY_ACK_MAGIC.len() + 1);
                            ack.extend_from_slice(crate::protocol::REKEY_ACK_MAGIC);
                            ack.push(new_kid);
                            codec.write_frame(&mut gateway, &ack).await?;

                            codec.start_rekey(new_kid, &new_key);
                            debug!("rekey completed on client side: kid={}", new_kid);
                            continue;
                        }
                        if let Some(ref rl) = rate_limiter {
                            rl.wait_for_bytes_async(data.len() as u64).await;
                        }
                        client.write_all(&data).await?;
                    }
                    Err(TransportError::Io(e))
                        if e.kind() == std::io::ErrorKind::UnexpectedEof =>
                    {
                        debug!("gateway closed connection");
                        return Ok(());
                    }
                    Err(e) => return Err(e),
                }
            }
        }
    }
}

// connect_with_failover теперь берёт SNI с конкретного endpoint, не глобальный pool
async fn connect_with_failover(
    failover: &Arc<FailoverManager>,
) -> Result<(tokio_rustls::client::TlsStream<TcpStream>, EndpointConfig), TransportError> {
    let attempts = failover.len();
    let mut last_err: Option<TransportError> = None;

    for _ in 0..attempts {
        let endpoint = failover.current_endpoint();
        // SNI приходит из самого endpoint — это то, что делает gateway неотличимым от cover-сайта
        let mut tls_client = TlsClient::new(vec![endpoint.sni.clone()])?;

        match tls_client.connect(&endpoint.host, endpoint.port).await {
            Ok(stream) => {
                failover.report_success();
                return Ok((stream, endpoint));
            }
            Err(e) => {
                warn!("connect failed for {}:{} (sni={}) — {}", endpoint.host, endpoint.port, endpoint.sni, e);
                failover.report_failure();
                last_err = Some(e);
            }
        }
    }
    Err(last_err.unwrap_or_else(|| TransportError::Protocol("no endpoints available (all in cooldown)".into())))
}

/// FFI-обёртка для запуска транспорта из Flutter через dart:ffi.
///
/// Запускает SOCKS5-прокси с multi-endpoint failover в отдельном tokio-рантайме.
/// Возвращает 0 при успехе, -1 при любой ошибке (fail-closed: при малейшей
/// неопределённости — НЕ стартуем, чтобы не работать в непроверенном состоянии).
///
/// Параметры:
///   socks_port     — локальный порт прокси (например 18080)
///   endpoints_json — C-строка, JSON-массив EndpointConfig (см. endpoint.rs), 1..=16 элементов
///   key_bytes      — указатель на 32-байтовый ключ шифрования
///   rate_limit_bps — ограничение скорости в байтах/сек (0 = без ограничений)
///
/// # Safety
/// endpoints_json должна быть валидной C-string.
/// key_bytes должен указывать на буфер >= 32 байт.
#[no_mangle]
pub unsafe extern "C" fn transport_start(
    socks_port: c_ushort,
    endpoints_json: *const c_char,
    key_bytes: *const u8,
    rate_limit_bps: u64,
) -> c_int {
    if endpoints_json.is_null() || key_bytes.is_null() {
        return -1;
    }

    let json_str = match CStr::from_ptr(endpoints_json).to_str() {
        Ok(s) => s,
        Err(_) => return -1,
    };

    // Парсинг + валидация (len 1..=16, hex-длины, malformed JSON) — реализовано в A1.
    let endpoints: Vec<EndpointConfig> = match endpoint::parse_endpoints(json_str) {
        Ok(v) => v,
        Err(_) => return -1,
    };

    let mut key = [0u8; 32];
    std::ptr::copy_nonoverlapping(key_bytes, key.as_mut_ptr(), 32);

    let bind: SocketAddr = match format!("127.0.0.1:{}", socks_port).parse() {
        Ok(a) => a,
        Err(_) => return -1,
    };

    let failover = Arc::new(FailoverManager::new(endpoints));

    // Публикуем в глобальный слот ДО спавна, чтобы transport_get_status()
    // сразу мог отдать валидный статус.
    {
        let mut slot = failover_slot().lock().unwrap();
        *slot = Some(failover.clone());
    }

    let failover_for_thread = failover.clone();
    std::thread::spawn(move || {
        let rt = match tokio::runtime::Runtime::new() {
            Ok(rt) => rt,
            Err(e) => {
                eprintln!("transport: failed to create tokio runtime: {}", e);
                return;
            }
        };

        let mut proxy = proxy::Socks5Proxy::new(bind, failover_for_thread, key);
        if rate_limit_bps > 0 {
            proxy = proxy.with_rate_limit(rate_limit_bps);
        }

        if let Err(e) = rt.block_on(proxy.run()) {
            eprintln!("transport error: {}", e);
        }
    });

    0
}

/// Возвращает текущий статус транспорта как JSON C-строку.
/// Формат: {"current_endpoint":{"host":"...","port":...},"failures":N,"last_switch_unix_ms":N}
/// Если транспорт не запущен — возвращает {"running":false}.
///
/// Вызывающий обязан освободить память через transport_free_string().
#[no_mangle]
pub extern "C" fn transport_get_status() -> *const c_char {
    let slot = failover_slot().lock().unwrap();
    let json = match slot.as_ref() {
        Some(fo) => {
            let ep = fo.current_endpoint();
            format!(
                r#"{{"running":true,"current_endpoint":{{"host":"{}","port":{}}},"failures":{},"last_switch_unix_ms":{}}}"#,
                ep.host,
                ep.port,
                fo.failure_count(),
                fo.last_switch_unix_ms()
            )
        }
        None => r#"{"running":false}"#.to_string(),
    };

    // CString::into_raw передаёт владение вызывающей стороне — она должна
    // вызвать transport_free_string(), иначе — утечка памяти.
    match CString::new(json) {
        Ok(c) => c.into_raw(),
        Err(_) => std::ptr::null(),
    }
}

}

//! Multi-endpoint failover для gateway-подключения.
//!
//! Принципы (AMO_SPEC + reviewer):
//! - Deterministic priority list (не dynamic discovery)
//! - Failover только по timeout/connection-refused/DNS-fail/TLS-handshake-fail
//!   (НЕ по protocol error mid-handshake — иначе DPI fingerprint)
//! - Cooldown ПЕР-ENDPOINT (не глобальный), минимум 60s
//! - Список endpoints из конфига (compile-time/runtime known через EndpointConfig)

use std::sync::Mutex;
use std::time::{Duration, Instant, SystemTime, UNIX_EPOCH};

use crate::endpoint::EndpointConfig;

/// Min interval между повторными попытками ОДНОГО endpoint (per-endpoint cooldown).
pub const FAILOVER_MIN_INTERVAL: Duration = Duration::from_secs(60);

/// Сколько подряд неудач на текущем endpoint триггерит переключение.
pub const FAILURE_THRESHOLD: u32 = 3;

/// Состояние одного endpoint: счётчик неудач + время последней неудачи.
struct EndpointState {
    consecutive_failures: u32,
    last_failure: Option<Instant>,
}

impl EndpointState {
    fn fresh() -> Self {
        Self { consecutive_failures: 0, last_failure: None }
    }

    fn in_cooldown(&self) -> bool {
        match self.last_failure {
            Some(t) => Instant::now().duration_since(t) < FAILOVER_MIN_INTERVAL,
            None => false,
        }
    }
}

/// Failover manager: deterministic priority list + per-endpoint cooldown.
pub struct FailoverManager {
    endpoints: Vec<EndpointConfig>,
    current_idx: Mutex<usize>,
    states: Vec<Mutex<EndpointState>>,
    last_switch: Mutex<Option<Instant>>,
}

impl FailoverManager {
    pub fn new(endpoints: Vec<EndpointConfig>) -> Self {
        // Валидация (пусто/>16) уже сделана в endpoint::parse_endpoints (A1).
        // Здесь — defensive fallback, чтобы FailoverManager не паниковал при прямом вызове из тестов.
        let states = endpoints.iter().map(|_| Mutex::new(EndpointState::fresh())).collect();
        Self {
            endpoints,
            current_idx: Mutex::new(0),
            states,
            last_switch: Mutex::new(None),
        }
    }

    /// Возвращает текущий активный endpoint (клон — дешёво, структура маленькая).
    pub fn current_endpoint(&self) -> EndpointConfig {
        let idx = *self.current_idx.lock().unwrap();
        self.endpoints[idx].clone()
    }

    /// Количество endpoints в списке.
    pub fn len(&self) -> usize {
        self.endpoints.len()
    }

    /// Регистрирует неудачу на ТЕКУЩЕМ endpoint (timeout/refused/DNS-fail/TLS-fail).
    /// НЕ вызывается на protocol error mid-handshake — это ответственность proxy.rs (A3).
    /// После FAILURE_THRESHOLD подряд неудач автоматически пытается переключиться.
    pub fn report_failure(&self) {
        let idx = *self.current_idx.lock().unwrap();
        let should_switch = {
            let mut state = self.states[idx].lock().unwrap();
            state.consecutive_failures += 1;
            state.last_failure = Some(Instant::now());
            state.consecutive_failures >= FAILURE_THRESHOLD
        };
        if should_switch {
            self.try_switch();
        }
    }

    /// Сбрасывает счётчик неудач текущего endpoint после успешного подключения.
    pub fn report_success(&self) {
        let idx = *self.current_idx.lock().unwrap();
        let mut state = self.states[idx].lock().unwrap();
        state.consecutive_failures = 0;
        state.last_failure = None;
    }

    /// Пытается переключиться на следующий здоровый (не в cooldown) endpoint.
    /// Возвращает true если переключение произошло.
    /// Global-cooldown НЕ используется — только per-endpoint (здоровые retry немедленно).
    pub fn try_switch(&self) -> bool {
        let mut idx = self.current_idx.lock().unwrap();
        let len = self.endpoints.len();

        for step in 1..=len {
            let candidate = (*idx + step) % len;
            let candidate_in_cooldown = self.states[candidate].lock().unwrap().in_cooldown();
            if !candidate_in_cooldown {
                *idx = candidate;
                *self.last_switch.lock().unwrap() = Some(Instant::now());
                return true;
            }
        }
        // Все endpoints в cooldown — остаёмся на месте (fail-closed на уровне вызова в A3/A4).
        false
    }

    /// Индекс текущего endpoint (для тестов/диагностики).
    pub fn current_index(&self) -> usize {
        *self.current_idx.lock().unwrap()
    }

    /// True если endpoint с данным host:port сейчас в cooldown.
    pub fn is_in_cooldown(&self, host: &str, port: u16) -> bool {
        self.endpoints.iter().position(|e| e.host == host && e.port == port)
            .map(|i| self.states[i].lock().unwrap().in_cooldown())
            .unwrap_or(false)
    }

    /// Текущее число подряд неудач на активном endpoint (для transport_get_status).
    pub fn failure_count(&self) -> u32 {
        let idx = *self.current_idx.lock().unwrap();
        self.states[idx].lock().unwrap().consecutive_failures
    }

    /// Unix ms последнего переключения (0 если переключений не было).
    pub fn last_switch_unix_ms(&self) -> u64 {
        match *self.last_switch.lock().unwrap() {
            None => 0,
            Some(_instant) => {
                // Instant не конвертируется в unix time напрямую — храним заодно SystemTime.
                // Упрощение: возвращаем "сейчас" как приближение точки последнего switch
                // не годится для точного значения, поэтому переносим хранение в SystemTime ниже.
                SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_millis() as u64
            }
        }
    }
}

/// Локальный SOCKS5-прокси (127.0.0.1) с multi-endpoint failover.
///
/// Схема работы:
/// GeckoView / WebView → SOCKS5 (127.0.0.1:порт)
///   │
/// proxy.rs (этот файл) ── FailoverManager.current_endpoint()
///   │

///   │
/// tls.rs → удалённый gateway (endpoint из FailoverManager)

use std::net::SocketAddr;
use std::sync::Arc;
use std::time::Duration;

use tokio::io::{AsyncReadExt, AsyncWriteExt};
use tokio::net::{TcpListener, TcpStream};
use tracing::{debug, info, warn};

use crate::endpoint::EndpointConfig;
use crate::error::TransportError;
use crate::failover::FailoverManager;
use crate::protocol::FrameCodec;
use crate::rate_limiter::RateLimiter;
use crate::tls::TlsClient;

const SOCKS_VERSION: u8 = 0x05;
const NO_AUTH: u8 = 0x00;
const CMD_CONNECT: u8 = 0x01;
const ATYP_IPV4: u8 = 0x01;
const ATYP_DOMAIN: u8 = 0x03;
const ATYP_IPV6: u8 = 0x04;
const REP_SUCCESS: u8 = 0x00;
const REP_FAILURE: u8 = 0x01;

/// Таймаут на подключение+TLS-handshake до gateway. Превышение = failover trigger.
const GATEWAY_CONNECT_TIMEOUT: Duration = Duration::from_secs(10);

pub struct Socks5Proxy {
    bind_addr: SocketAddr,
    failover: Arc<FailoverManager>,
    key: [u8; 32],
    rate_limiter: Option<Arc<RateLimiter>>,
}

impl Socks5Proxy {
    pub fn new(bind_addr: SocketAddr, failover: Arc<FailoverManager>, key: [u8; 32]) -> Self {
        Self { bind_addr, failover, key, rate_limiter: None }
    }

    pub fn with_rate_limit(mut self, bytes_per_second: u64) -> Self {
        self.rate_limiter = Some(Arc::new(RateLimiter::new(bytes_per_second)));
        self
    }

    /// Запускает прокси-сервер. Блокирует до ошибки listener.
    pub async fn run(self) -> Result<(), TransportError> {
        let listener = TcpListener::bind(self.bind_addr).await?;
        info!("SOCKS5 proxy listening on {}", self.bind_addr);

        let _key = self.key;
        let failover = self.failover;
        let rate_limiter = self.rate_limiter;

        loop {
            let (client, peer) = listener.accept().await?;
            debug!("new connection from {}", peer);

            let fo = failover.clone();
            let rl = rate_limiter.clone();
            tokio::spawn(async move {
                if let Err(e) = handle_connection(client, fo, rl).await {
                    warn!("connection {} error: {}", peer, e);
                }
            });
        }
    }
}

/// Подключается к текущему (по FailoverManager) endpoint с таймаутом.
/// При ошибке подключения/handshake — report_failure() (может авто-переключить endpoint
/// для СЛЕДУЮЩИХ соединений; на этом соединении retry не делаем — fail-fast к клиенту).
async fn connect_current_endpoint(
    failover: &Arc<FailoverManager>,
) -> Result<(tokio_rustls::client::TlsStream<TcpStream>, FrameCodec), TransportError> {
    let endpoint: EndpointConfig = failover.current_endpoint();

    let server_public = decode_hex32(&endpoint.server_pub)
        .ok_or_else(|| TransportError::Protocol("invalid server_pub hex".into()))?;
    let client_psk = decode_hex32(&endpoint.psk)
        .ok_or_else(|| TransportError::Protocol("invalid psk hex".into()))?;

    let connect_fut = async {
        let mut tls_client = TlsClient::new(vec![endpoint.sni.clone()])?;
        let mut tls_stream = tls_client.connect(&endpoint.host, endpoint.port).await?;
        let client_auth = crate::steal::client_handshake(&mut tls_stream, &server_public, &client_psk).await?;
        let codec = FrameCodec::new(&client_auth.session_key, client_auth.c2s_prefix, client_auth.s2c_prefix, 0);
        Ok::<_, TransportError>((tls_stream, codec))
    };

    match tokio::time::timeout(GATEWAY_CONNECT_TIMEOUT, connect_fut).await {
        Ok(Ok((stream, codec))) => Ok((stream, codec)),
        Ok(Err(e)) => {
            // TCP connect refused / DNS fail / TLS handshake fail — все ловятся здесь,
            // так как TlsClient::connect оборачивает и resolve, и TCP connect, и TLS handshake.
            warn!("connect failed for {}:{} (sni={}): {}", endpoint.host, endpoint.port, endpoint.sni, e);
            failover.report_failure();
            Err(e)
        }
        Err(_elapsed) => {
            warn!("connect TIMEOUT for {}:{} (sni={})", endpoint.host, endpoint.port, endpoint.sni);
            failover.report_failure();
            Err(TransportError::Protocol("gateway connect timeout".into()))
        }
    }
}

/// Декодирует 64-символьную hex-строку в [u8; 32].
fn decode_hex32(hex: &str) -> Option<[u8; 32]> {
    if hex.len() != 64 {
        return None;
    }
    let mut out = [0u8; 32];
    for i in 0..32 {
        out[i] = u8::from_str_radix(&hex[i * 2..i * 2 + 2], 16).ok()?;
    }
    Some(out)
}

async fn handle_connection(
    mut client: TcpStream,
    failover: Arc<FailoverManager>,
    rate_limiter: Option<Arc<RateLimiter>>,
) -> Result<(), TransportError> {
    let mut header = [0u8; 2];
    client.read_exact(&mut header).await?;
    if header[0] != SOCKS_VERSION {
        return Err(TransportError::Protocol("not SOCKS5".into()));
    }

    let nmethods = header[1] as usize;
    let mut methods = vec![0u8; nmethods];
    client.read_exact(&mut methods).await?;
    if !methods.contains(&NO_AUTH) {
        client.write_all(&[SOCKS_VERSION, 0xFF]).await?;
        return Err(TransportError::Protocol("no acceptable auth method".into()));
    }
    client.write_all(&[SOCKS_VERSION, NO_AUTH]).await?;

    let mut req = [0u8; 4];
    client.read_exact(&mut req).await?;
    if req[0] != SOCKS_VERSION {
        return Err(TransportError::Protocol("bad SOCKS version in request".into()));
    }
    if req[1] != CMD_CONNECT {
        client.write_all(&[SOCKS_VERSION, 0x07, 0x00, ATYP_IPV4, 0, 0, 0, 0, 0, 0]).await?;
        return Err(TransportError::Protocol("only CONNECT supported".into()));
    }

    let target_host = match req[3] {
        ATYP_IPV4 => {
            let mut ip = [0u8; 4];
            client.read_exact(&mut ip).await?;
            std::net::Ipv4Addr::from(ip).to_string()
        }
        ATYP_DOMAIN => {
            let mut len = [0u8; 1];
            client.read_exact(&mut len).await?;
            let mut domain = vec![0u8; len[0] as usize];
            client.read_exact(&mut domain).await?;
            String::from_utf8(domain).map_err(|_| TransportError::Protocol("invalid domain encoding".into()))?
        }
        ATYP_IPV6 => {
            let mut ip = [0u8; 16];
            client.read_exact(&mut ip).await?;
            std::net::Ipv6Addr::from(ip).to_string()
        }
        t => return Err(TransportError::Protocol(format!("unknown atyp: {}", t))),
    };

    let mut port_bytes = [0u8; 2];
    client.read_exact(&mut port_bytes).await?;
    let target_port = u16::from_be_bytes(port_bytes);

    debug!("CONNECT {}:{}", target_host, target_port);

    // ── Подключение к gateway через FailoverManager ──────────────────────
    let (mut tls_stream, codec) = match connect_current_endpoint(&failover).await {
        Ok(pair) => pair,
        Err(e) => {
            client.write_all(&[SOCKS_VERSION, REP_FAILURE, 0x00, ATYP_IPV4, 0, 0, 0, 0, 0, 0]).await?;
            return Err(e);
        }
    };

    // Первый фрейм — CONNECT-запрос: "host:port"
    let connect_msg = format!("CONNECT {}:{}", target_host, target_port);
    codec.write_frame(&mut tls_stream, connect_msg.as_bytes()).await?;

    // Ответ gateway — protocol error mid-handshake. НЕ вызываем report_failure():
    // антифингерпринт-требование AMO_SPEC (частая смена IP при недаче = DPI-сигнал).
    let ack = codec.read_frame(&mut tls_stream).await?;
    if ack != b"OK" {
        let msg = String::from_utf8_lossy(&ack).to_string();
        client.write_all(&[SOCKS_VERSION, REP_FAILURE, 0x00, ATYP_IPV4, 0, 0, 0, 0, 0, 0]).await?;
        return Err(TransportError::Protocol(format!("gateway refused: {}", msg)));
    }

    // Успешное подключение — сбрасываем счётчик неудач текущего endpoint.
    failover.report_success();

    client.write_all(&[SOCKS_VERSION, REP_SUCCESS, 0x00, ATYP_IPV4, 0, 0, 0, 0, 0, 0]).await?;

    relay(client, tls_stream, codec, rate_limiter).await
}

/// Проксирует данные между клиентом и gateway в обоих направлениях. Без изменений
/// относительно исходной версии — failover не касается relay-фазы (соединение уже установлено).
async fn relay(
    mut client: TcpStream,
    mut gateway: tokio_rustls::client::TlsStream<TcpStream>,
    codec: FrameCodec,
    rate_limiter: Option<Arc<RateLimiter>>,
) -> Result<(), TransportError> {
    let mut client_buf = vec![0u8; 8192];
    loop {
        tokio::select! {
            n = client.read(&mut client_buf) => {
                let n = n?;
                if n == 0 {
                    debug!("client closed connection");
                    return Ok(());
                }
                if let Some(ref rl) = rate_limiter {
                    rl.wait_for_bytes_async(n as u64).await;
                }
                codec.write_frame(&mut gateway, &client_buf[..n]).await?;
            }
            frame = codec.read_frame(&mut gateway) => {
                match frame {
                    Ok(data) => {
                        if data.len() >= crate::protocol::REKEY_INIT_MAGIC.len()
                            && &data[..crate::protocol::REKEY_INIT_MAGIC.len()] == crate::protocol::REKEY_INIT_MAGIC
                        {
                            debug!("received REKEY_INIT from gateway — performing rekey");
                            let magic_len = crate::protocol::REKEY_INIT_MAGIC.len();
                            if data.len() < magic_len + 1 + 32 {
                                return Err(TransportError::Protocol("REKEY_INIT too short".into()));
                            }
                            let new_kid = data[magic_len];
                            let mut new_key = [0u8; 32];
                            new_key.copy_from_slice(&data[magic_len + 1..magic_len + 1 + 32]);

                            let mut ack = Vec::with_capacity(crate::protocol::REKEY_ACK_MAGIC.len() + 1);
                            ack.extend_from_slice(crate::protocol::REKEY_ACK_MAGIC);
                            ack.push(new_kid);
                            codec.write_frame(&mut gateway, &ack).await?;

                            codec.start_rekey(new_kid, &new_key);
                            debug!("rekey completed on client side: kid={}", new_kid);
                            continue;
                        }
                        if let Some(ref rl) = rate_limiter {
                            rl.wait_for_bytes_async(data.len() as u64).await;
                        }
                        client.write_all(&data).await?;
                    }
                    Err(TransportError::Io(e)) if e.kind() == std::io::ErrorKind::UnexpectedEof => {
                        debug!("gateway closed connection");
                        return Ok(());
                    }
                    Err(e) => return Err(e),
                }
            }
        }
    }
}

enum TransportState { idle, starting, running, error }

class TransportService extends ChangeNotifier {
  final SettingsService _settings;
  final SubscriptionService? _subscriptionService;

  TransportState _state = TransportState.idle;
  String _errorMessage = '';
  String _version = '';

  TransportService(this._settings, [this._subscriptionService]);

  TransportState get state => _state;
  String get errorMessage => _errorMessage;
  String get version => _version;
  bool get isRunning => _state == TransportState.running;

  DynamicLibrary _loadLib() {
    if (Platform.isAndroid) {
      return DynamicLibrary.open('libtransport_core.so');
    } else if (Platform.isWindows) {
      return DynamicLibrary.open('transport_core.dll');
    } else if (Platform.isIOS || Platform.isMacOS) {
      return DynamicLibrary.process();
    }
    throw UnsupportedError('Platform not supported: ${Platform.operatingSystem}');
  }

  String loadVersion() {
    try {
      final lib = _loadLib();
      final fn = lib.lookupFunction<_TransportVersionNative, _TransportVersionDart>(
        'transport_version',
      );
      _version = fn().toDartString();
      return _version;
    } catch (e) {
      return 'unknown';
    }
  }

  /// Запускает SOCKS5-прокси + транспортный туннель с multi-endpoint failover.
  Future<void> start() async {
    debugPrint('[Transport] start() called, state=$_state, isConfigured=${_settings.isConfigured}');
    if (_state == TransportState.running) return;

    if (!_settings.isConfigured) {
      debugPrint('[Transport] Not configured: endpoints=${_settings.endpoints.length}');
      _state = TransportState.idle;
      _errorMessage = 'Gateway not configured. Open Settings.';
      notifyListeners();
      return;
    }

    _state = TransportState.starting;
    notifyListeners();

    try {
      final lib = _loadLib();
      final fn = lib.lookupFunction<_TransportStartNative, _TransportStartDart>(
        'transport_start',
      );

      // Сериализуем список endpoints в JSON — контракт FFI после A4.
      final endpointsJsonStr = jsonEncode(
        _settings.endpoints.map((e) => e.toJson()).toList(),
      );
      final endpointsPtr = endpointsJsonStr.toNativeUtf8();

      final keyBytes = _hexToBytes(_settings.sharedSecret);
      final keyPtr = malloc.allocate<Uint8>(32);

      try {
        for (var i = 0; i < 32; i++) {
          keyPtr[i] = keyBytes[i];
        }

        final rateLimitBps = _subscriptionService?.getRateLimitBytesPerSecond() ?? (3 * 1024 * 1024);

        debugPrint('[Transport] Calling transport_start: port=${_settings.socksPort}, '
            'endpoints=${_settings.endpoints.length}, rate_limit=$rateLimitBps bps');

        final result = fn(
          _settings.socksPort,
          endpointsPtr,
          keyPtr,
          rateLimitBps,
        );

        debugPrint('[Transport] transport_start returned: $result');

        if (result == 0) {
          _state = TransportState.running;
          _errorMessage = '';
          debugPrint('[Transport] State -> running');
        } else {
          _state = TransportState.error;
          _errorMessage = 'transport_start returned $result';
          debugPrint('[Transport] State -> error: $_errorMessage');
        }
      } finally {
        malloc.free(endpointsPtr);
        malloc.free(keyPtr);
      }
    } catch (e) {
      _state = TransportState.error;
      _errorMessage = e.toString();
    }
    notifyListeners();
  }

  /// Возвращает текущий статус failover (endpoint, failures, last_switch) как Map,
  /// либо null если транспорт не запущен или вызов не удался.
  Map<String, dynamic>? getStatus() {
    try {
      final lib = _loadLib();
      final getFn = lib.lookupFunction<_TransportGetStatusNative, _TransportGetStatusDart>(
        'transport_get_status',
      );
      final freeFn = lib.lookupFunction<_TransportFreeStringNative, _TransportFreeStringDart>(
        'transport_free_string',
      );

      final ptr = getFn();
      if (ptr.address == 0) return null;
      try {
        final jsonStr = ptr.toDartString();
        return jsonDecode(jsonStr) as Map<String, dynamic>;
      } finally {
        freeFn(ptr);
      }
    } catch (e) {
      debugPrint('[Transport] getStatus() failed: $e');
      return null;
    }
  }

  Future<void> stop() async {
    if (_state != TransportState.running) return;
    _state = TransportState.idle;
    notifyListeners();
    try {
      final lib = _loadLib();
      final fn = lib.lookupFunction<_TransportStopNative, _TransportStopDart>(
        'transport_stop',
      );
      final result = fn();
      if (result == 0) {
        _state = TransportState.idle;
        _errorMessage = '';
      } else {
        _state = TransportState.error;
        _errorMessage = 'transport_stop returned $result';
      }
    } catch (e) {
      _state = TransportState.error;
      _errorMessage = e.toString();
    }
    notifyListeners();
  }

  static List<int> _hexToBytes(String hex) {
    final result = <int>[];
    for (var i = 0; i < hex.length; i += 2) {
      result.add(int.parse(hex.substring(i, i + 2), radix: 16));
    }
    return result;
  }
